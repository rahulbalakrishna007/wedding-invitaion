<script>
    // 1. SET THE WEDDING DATE HERE
    const weddingDate = new Date("May 18, 2025 10:00:00").getTime();

    // 2. Countdown Logic
    const timer = setInterval(function() {
        const now = new Date().getTime();
        const distance = weddingDate - now;

        const days = Math.floor(distance / (1000 * 60 * 60 * 24));
        const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
        const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
        const seconds = Math.floor((distance % (1000 * 60)) / 1000);

        document.getElementById("days").innerText = days;
        document.getElementById("hours").innerText = hours;
        document.getElementById("minutes").innerText = minutes;
        document.getElementById("seconds").innerText = seconds;

        if (distance < 0) {
            clearInterval(timer);
            document.getElementById("countdown").innerHTML = "<h3>The Celebration has Begun!</h3>";
        }
    }, 1000);

    // 3. WhatsApp Sharing Logic
    function shareWhatsapp() {
        const text = encodeURIComponent("You're invited! Join us for the wedding of Saranya & Ajesh. View the invitation here: " + window.location.href);
        window.open(https://wa.me/?text=${text}, '_blank');
    }
</script>
